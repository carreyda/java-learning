package com.example.blogapi.config;

                        .scheme("bearer")
                        .bearerFormat("JWT")
                        .description("在此输入 JWT Token（不需要加 Bearer 前缀）")));
    }
}

## 2. Swagger 注解详解

### 2.1 Controller 级别

```java
@Tag(name = "文章管理", description = "文章的增删改查及互动接口")
@RestController
@RequestMapping("/api/v1/articles")
public class ArticleController {

    @Operation(
        summary = "获取文章列表",
        description = "支持分页、关键词搜索、状态过滤。未登录用户只能看到已发布的文章。"
    )
    @ApiResponses({
        @ApiResponse(responseCode = "200", description = "查询成功"),
        @ApiResponse(responseCode = "400", description = "参数格式错误")
    })
    @GetMapping
    public Result<PageResult<ArticleVO>> list(
        @Parameter(description = "页码，从1开始", example = "1")
        @RequestParam(defaultValue = "1") Integer page,

        @Parameter(description = "每页数量，最大100", example = "10")
        @RequestParam(defaultValue = "10") Integer size,

        @Parameter(description = "关键词（搜索标题）")
        @RequestParam(required = false) String keyword,

        @Parameter(description = "文章状态", schema = @Schema(allowableValues = {"DRAFT", "PUBLISHED", "ARCHIVED"}))
        @RequestParam(required = false) String status
    ) {
        return Result.ok(articleService.listArticles(page, size, keyword, status));
    }

    @Operation(summary = "获取文章详情")
    @GetMapping("/{id}")
    public Result<ArticleDetailVO> getById(
        @Parameter(description = "文章ID", required = true, example = "1")
        @PathVariable Long id
    ) {
        return Result.ok(articleService.getArticleDetail(id, null));
    }

    @Operation(summary = "创建文章", description = "需要登录。创建后文章状态为草稿。")
    @SecurityRequirement(name = "Bearer Token")  // 该接口需要认证
    @PostMapping
    public Result<ArticleVO> create(
        @io.swagger.v3.oas.annotations.parameters.RequestBody(
            description = "文章创建参数",
            required = true
        )
        @RequestBody @Valid CreateArticleDTO dto
    ) {
        return Result.ok(articleService.create(dto));
    }
}
```

## 4. Service 单元测试

### 4.1 用 Mockito Mock 依赖

```java
package com.example.blogapi.service;

import com.example.blogapi.common.AppException;
import com.example.blogapi.mapper.UserMapper;
import com.example.blogapi.model.dto.CreateUserDTO;
import com.example.blogapi.model.entity.User;
import com.example.blogapi.model.vo.UserVO;
import com.example.blogapi.service.impl.UserServiceImpl;
import org.junit.jupiter.api.*;
import org.junit.jupiter.api.extension.ExtendWith;
import org.mockito.*;
import org.mockito.junit.jupiter.MockitoExtension;

import static org.assertj.core.api.Assertions.*;
import static org.mockito.BDDMockito.*;

@ExtendWith(MockitoExtension.class)
@DisplayName("UserService 单元测试")
class UserServiceTest {

    @Mock
    private UserMapper userMapper;  // Mock 数据库操作

    @InjectMocks
    private UserServiceImpl userService;  // 注入 Mock 的 UserServiceImpl

    @Nested
    @DisplayName("createUser 方法")
    class CreateUserTest {

        private CreateUserDTO validDTO;

        @BeforeEach
        void setUp() {
            validDTO = new CreateUserDTO();
            validDTO.setUsername("testuser");
            validDTO.setEmail("test@example.com");
            validDTO.setPassword("password123");
        }

        @Test
        @DisplayName("正常创建用户应该成功")
        void createUser_ShouldSucceed_WhenValidInput() {
            // Given（准备数据）
            given(userMapper.selectCount(any())).willReturn(0L);  // 用户名不存在
            given(userMapper.insert(any(User.class))).willReturn(1);

            // When（执行被测方法）
            UserVO result = userService.createUser(validDTO);

            // Then（验证结果）
            assertThat(result).isNotNull();
            assertThat(result.getUsername()).isEqualTo("testuser");
            assertThat(result.getEmail()).isEqualTo("test@example.com");

            // 验证 userMapper.insert 被调用了一次
            verify(userMapper, times(1)).insert(any(User.class));
        }

        @Test
        @DisplayName("用户名已存在时应该抛出 AppException")
        void createUser_ShouldThrow_WhenUsernameExists() {
            // Given
            given(userMapper.selectCount(any())).willReturn(1L);  // 用户名已存在

            // Then（验证抛出异常）
            assertThatThrownBy(() -> userService.createUser(validDTO))
                .isInstanceOf(AppException.class)
                .hasMessageContaining("testuser");

            // 验证 insert 没有被调用
            verify(userMapper, never()).insert(any());
        }
    }

    @Nested
    @DisplayName("getUserVO 方法")
    class GetUserVOTest {

        @Test
        @DisplayName("用户存在时应该返回 VO")
        void getUserVO_ShouldReturnVO_WhenUserExists() {
            // Given
            User mockUser = new User();
            mockUser.setId(1L);
            mockUser.setUsername("testuser");
            mockUser.setEmail("test@example.com");
            given(userMapper.selectById(1L)).willReturn(mockUser);

            // When
            UserVO result = userService.getUserVO(1L);

            // Then
            assertThat(result.getId()).isEqualTo(1L);
            assertThat(result.getUsername()).isEqualTo("testuser");
        }

        @Test
        @DisplayName("用户不存在时应该抛出 AppException")
        void getUserVO_ShouldThrow_WhenUserNotFound() {
            // Given
            given(userMapper.selectById(anyLong())).willReturn(null);

            // Then
            assertThatThrownBy(() -> userService.getUserVO(999L))
                .isInstanceOf(AppException.class);
        }
    }
}
```

## 5. Controller 集成测试

### 5.1 MockMvc 测试（不启动真实服务器）

```java
package com.example.blogapi.controller;

import com.example.blogapi.common.Result;
import com.example.blogapi.model.dto.CreateUserDTO;
import com.example.blogapi.model.vo.UserVO;
import com.example.blogapi.service.UserService;
import com.fasterxml.jackson.databind.ObjectMapper;
import org.junit.jupiter.api.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.autoconfigure.web.servlet.WebMvcTest;
import org.springframework.boot.test.mock.mockito.MockBean;
import org.springframework.http.MediaType;
import org.springframework.test.web.servlet.MockMvc;

import static org.mockito.BDDMockito.*;
import static org.springframework.test.web.servlet.request.MockMvcRequestBuilders.*;
import static org.springframework.test.web.servlet.result.MockMvcResultMatchers.*;
import static org.springframework.test.web.servlet.result.MockMvcResultHandlers.*;

@WebMvcTest(UserController.class)  // 只启动 UserController 相关的 Bean
@DisplayName("UserController 集成测试")
class UserControllerTest {

    @Autowired
    private MockMvc mockMvc;

    @Autowired
    private ObjectMapper objectMapper;

    @MockBean  // 在 Spring 上下文中 Mock UserService
    private UserService userService;

    @Test
    @DisplayName("GET /api/users/{id} - 用户存在时返回 200")
    void getById_ShouldReturn200_WhenUserExists() throws Exception {
        // Given
        UserVO mockVO = new UserVO();
        mockVO.setId(1L);
        mockVO.setUsername("testuser");
        given(userService.getUserVO(1L)).willReturn(mockVO);

        // When & Then
        mockMvc.perform(get("/api/users/1")
                .accept(MediaType.APPLICATION_JSON))
            .andDo(print())  // 打印请求/响应详情
            .andExpect(status().isOk())
            .andExpect(content().contentType(MediaType.APPLICATION_JSON))
            .andExpect(jsonPath("$.code").value(200))
            .andExpect(jsonPath("$.data.id").value(1))
            .andExpect(jsonPath("$.data.username").value("testuser"));
    }

    @Test
    @DisplayName("GET /api/users/{id} - 用户不存在时返回 404")
    void getById_ShouldReturn404_WhenUserNotFound() throws Exception {
        // Given
        given(userService.getUserVO(999L))
            .willThrow(new AppException(ErrorCode.USER_NOT_FOUND));

        // When & Then
        mockMvc.perform(get("/api/users/999"))
            .andExpect(status().isNotFound())
            .andExpect(jsonPath("$.code").value(100101));
    }

    @Test
    @DisplayName("POST /api/users - 参数合法时创建成功返回 201")
    void create_ShouldReturn201_WhenValidInput() throws Exception {
        // Given
        CreateUserDTO dto = new CreateUserDTO();
        dto.setUsername("newuser");
        dto.setEmail("new@example.com");
        dto.setPassword("password123");

        UserVO mockVO = new UserVO();
        mockVO.setId(5L);
        mockVO.setUsername("newuser");
        given(userService.createUser(any(CreateUserDTO.class))).willReturn(mockVO);

        // When & Then
        mockMvc.perform(post("/api/users")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(dto)))
            .andExpect(status().isCreated())  // 201
            .andExpect(jsonPath("$.data.id").value(5))
            .andExpect(jsonPath("$.data.username").value("newuser"));
    }

    @Test
    @DisplayName("POST /api/users - 用户名为空时返回 400")
    void create_ShouldReturn400_WhenUsernameBlank() throws Exception {
        // Given：构造无效 DTO
        CreateUserDTO invalidDTO = new CreateUserDTO();
        invalidDTO.setUsername("");  // 空用户名
        invalidDTO.setEmail("test@example.com");
        invalidDTO.setPassword("password123");

        // When & Then
        mockMvc.perform(post("/api/users")
                .contentType(MediaType.APPLICATION_JSON)
                .content(objectMapper.writeValueAsString(invalidDTO)))
            .andExpect(status().isBadRequest())
            .andExpect(jsonPath("$.code").value(400))
            .andExpect(jsonPath("$.data.username").exists());  // 返回字段错误信息
    }

    @Test
    @DisplayName("DELETE /api/users/{id} - 删除成功返回 204")
    void delete_ShouldReturn204_WhenSuccess() throws Exception {
        willDoNothing().given(userService).deleteUser(1L);

        mockMvc.perform(delete("/api/users/1"))
            .andExpect(status().isNoContent());
    }
}
```

## 常见问题

**Q：`@WebMvcTest` 和 `@SpringBootTest` 哪个更适合 Controller 测试？**  
A：`@WebMvcTest` 只启动 Web 层（速度快），配合 `@MockBean` 模拟 Service，适合测试 Controller 的请求解析、参数校验、响应格式。`@SpringBootTest` 启动完整应用（速度慢），适合端到端集成测试。Controller 的单元测试推荐用 `@WebMvcTest`。

**Q：测试数据库用 H2 还是真实 MySQL？**  
A：单元/集成测试用 H2（速度快、无需启动额外服务、测试间隔离）。端到端测试和性能测试用真实 MySQL（行为更接近生产）。H2 的 `MODE=MySQL` 模式能兼容大部分 MySQL 语法，但有细微差别，复杂 SQL 需要测试验证。

**Q：Mockito 的 `@Mock` 和 `@MockBean` 有什么区别？**  
A：`@Mock` 是纯 Mockito 注解，不涉及 Spring，配合 `@ExtendWith(MockitoExtension.class)` 使用，速度最快。`@MockBean` 是 Spring 注解，会在 Spring 上下文中替换对应的 Bean，配合 `@WebMvcTest` 或 `@SpringBootTest` 使用。
