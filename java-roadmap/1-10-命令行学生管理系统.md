### Student.java

```java
package model;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

public class Student {
    private String studentId;       // 学号（自动生成）
    private String name;            // 姓名
    private int age;                // 年龄
    private String gender;          // 性别
    private String major;           // 专业
    private LocalDate enrollDate;   // 入学日期
    private List<Grade> grades;     // 成绩列表

    public Student(String studentId, String name, int age,
                   String gender, String major) {
        this.studentId = studentId;
        this.name = name;
        this.age = age;
        this.gender = gender;
        this.major = major;
        this.enrollDate = LocalDate.now();
        this.grades = new ArrayList<>();
    }

    // Getters
    public String getStudentId() { return studentId; }
    public String getName() { return name; }
    public int getAge() { return age; }
    public String getGender() { return gender; }
    public String getMajor() { return major; }
    public LocalDate getEnrollDate() { return enrollDate; }
    public List<Grade> getGrades() { return grades; }

    // Setters（只允许修改部分字段）
    public void setName(String name) { this.name = name; }
    public void setAge(int age) { this.age = age; }
    public void setMajor(String major) { this.major = major; }

    // 添加成绩
    public void addGrade(Grade grade) {
        // 如果已有该课程成绩，替换
        grades.removeIf(g -> g.getCourse().equals(grade.getCourse()));
        grades.add(grade);
    }

    // 计算平均分
    public double getAverageScore() {
        if (grades.isEmpty()) return 0.0;
        return grades.stream()
            .mapToDouble(Grade::getScore)
            .average()
            .orElse(0.0);
    }

    @Override
    public String toString() {
        return String.format("%-10s %-8s %3d岁  %-4s  %-12s  入学：%s",
            studentId, name, age, gender, major, enrollDate);
    }
}
```

### Grade.java

```java
package model;

public class Grade {
    private Course course;
    private double score;
    private String level;  // 优秀/良好/及格/不及格

    public Grade(Course course, double score) {
        if (score < 0 || score > 100) {
            throw new IllegalArgumentException("成绩必须在 0-100 之间：" + score);
        }
        this.course = course;
        this.score = score;
        this.level = calcLevel(score);
    }

    private String calcLevel(double score) {
        if (score >= 90) return "优秀";
        if (score >= 75) return "良好";
        if (score >= 60) return "及格";
        return "不及格";
    }

    public Course getCourse() { return course; }
    public double getScore() { return score; }
    public String getLevel() { return level; }

    @Override
    public String toString() {
        return String.format("%-12s  %.1f分  %s",
            course.getCourseName(), score, level);
    }
}
```

### InMemoryStudentRepository.java

```java
package repository;

import model.Student;

import java.util.*;
import java.util.stream.Collectors;

public class InMemoryStudentRepository implements StudentRepository {

    // 用 LinkedHashMap 保持插入顺序
    private final Map<String, Student> store = new LinkedHashMap<>();

    @Override
    public void save(Student student) {
        store.put(student.getStudentId(), student);
    }

    @Override
    public Optional<Student> findById(String studentId) {
        return Optional.ofNullable(store.get(studentId));
    }

    @Override
    public List<Student> findByName(String name) {
        return store.values().stream()
            .filter(s -> s.getName().contains(name))
            .collect(Collectors.toList());
    }

    @Override
    public List<Student> findAll() {
        return new ArrayList<>(store.values());
    }

    @Override
    public void update(Student student) {
        store.put(student.getStudentId(), student);
    }

    @Override
    public boolean deleteById(String studentId) {
        return store.remove(studentId) != null;
    }

    @Override
    public boolean existsByName(String name) {
        return store.values().stream()
            .anyMatch(s -> s.getName().equals(name));
    }

    @Override
    public int count() {
        return store.size();
    }
}
```

## 7. 业务层实现

### StudentService.java

```java
package service;

import exception.DuplicateStudentException;
import exception.StudentNotFoundException;
import model.Course;
import model.Grade;
import model.Student;
import repository.StudentRepository;

import java.util.*;
import java.util.stream.Collectors;

public class StudentService {

    private final StudentRepository repository;

    // 预定义课程
    private static final Map<String, Course> COURSES = new LinkedHashMap<>();

    static {
        COURSES.put("C001", new Course("C001", "高等数学", 4));
        COURSES.put("C002", new Course("C002", "大学英语", 3));
        COURSES.put("C003", new Course("C003", "Java程序设计", 4));
        COURSES.put("C004", new Course("C004", "数据结构", 3));
        COURSES.put("C005", new Course("C005", "数据库原理", 3));
    }

    // 学号序列
    private int idSequence = 1;

    public StudentService(StudentRepository repository) {
        this.repository = repository;
    }

    // ===== 学生管理 =====

    public Student addStudent(String name, int age, String gender, String major) {
        // 校验年龄
        if (age < 15 || age > 60) {
            throw new IllegalArgumentException("年龄不合法：" + age);
        }
        // 校验姓名
        if (name == null || name.isBlank()) {
            throw new IllegalArgumentException("姓名不能为空");
        }
        // 检查重名
        if (repository.existsByName(name)) {
            throw new DuplicateStudentException(name);
        }

        String studentId = generateStudentId();
        Student student = new Student(studentId, name, age, gender, major);
        repository.save(student);
        return student;
    }

    public Student getStudent(String studentId) {
        return repository.findById(studentId)
            .orElseThrow(() -> new StudentNotFoundException(studentId));
    }

    public List<Student> searchByName(String name) {
        return repository.findByName(name);
    }

    public List<Student> getAllStudents() {
        return repository.findAll();
    }

    public void updateStudent(String studentId, String name, int age, String major) {
        Student student = getStudent(studentId);
        if (name != null && !name.isBlank()) student.setName(name);
        if (age > 0) student.setAge(age);
        if (major != null && !major.isBlank()) student.setMajor(major);
        repository.update(student);
    }

    public void deleteStudent(String studentId) {
        if (!repository.deleteById(studentId)) {
            throw new StudentNotFoundException(studentId);
        }
    }

    // ===== 成绩管理 =====

    public Map<String, Course> getCourses() {
        return Collections.unmodifiableMap(COURSES);
    }

    public void addGrade(String studentId, String courseId, double score) {
        Student student = getStudent(studentId);
        Course course = COURSES.get(courseId);
        if (course == null) {
            throw new IllegalArgumentException("课程不存在：" + courseId);
        }
        student.addGrade(new Grade(course, score));
        repository.update(student);
    }

    public List<Grade> getGrades(String studentId) {
        return getStudent(studentId).getGrades();
    }

    public double getAverageScore(String studentId) {
        return getStudent(studentId).getAverageScore();
    }

    // 某门课的成绩排名
    public List<Map.Entry<Student, Double>> getCourseRanking(String courseId) {
        Course course = COURSES.get(courseId);
        if (course == null) throw new IllegalArgumentException("课程不存在：" + courseId);

        Map<Student, Double> scoreMap = new LinkedHashMap<>();
        for (Student student : repository.findAll()) {
            student.getGrades().stream()
                .filter(g -> g.getCourse().equals(course))
                .findFirst()
                .ifPresent(g -> scoreMap.put(student, g.getScore()));
        }

        return scoreMap.entrySet().stream()
            .sorted((a, b) -> Double.compare(b.getValue(), a.getValue()))
            .collect(Collectors.toList());
    }

    // ===== 统计 =====

    // 各科平均分
    public Map<String, Double> getCourseAverages() {
        Map<String, Double> result = new LinkedHashMap<>();
        for (Course course : COURSES.values()) {
            List<Student> students = repository.findAll();
            OptionalDouble avg = students.stream()
                .flatMap(s -> s.getGrades().stream())
                .filter(g -> g.getCourse().equals(course))
                .mapToDouble(Grade::getScore)
                .average();
            if (avg.isPresent()) {
                result.put(course.getCourseName(), avg.getAsDouble());
            }
        }
        return result;
    }

    // 成绩等级分布统计
    public Map<String, Long> getGradeLevelStats() {
        List<Grade> allGrades = repository.findAll().stream()
            .flatMap(s -> s.getGrades().stream())
            .collect(Collectors.toList());

        Map<String, Long> stats = new LinkedHashMap<>();
        stats.put("优秀(≥90)", allGrades.stream().filter(g -> g.getScore() >= 90).count());
        stats.put("良好(75-89)", allGrades.stream().filter(g -> g.getScore() >= 75 && g.getScore() < 90).count());
        stats.put("及格(60-74)", allGrades.stream().filter(g -> g.getScore() >= 60 && g.getScore() < 75).count());
        stats.put("不及格(<60)", allGrades.stream().filter(g -> g.getScore() < 60).count());
        return stats;
    }

    // ===== 工具方法 =====

    private String generateStudentId() {
        return String.format("S%04d", idSequence++);
    }
}
```

```java
            List<Grade> grades = service.getGrades(studentId);
            System.out.println("\n" + student.getName() + " 的成绩单：");
            System.out.println("─".repeat(35));

                System.out.println("暂无成绩记录");
            } else {
                grades.forEach(System.out::println);
                System.out.println("─".repeat(35));
                System.out.printf("平均分：%.2f%n", service.getAverageScore(studentId));
            }
        } catch (BusinessException e) {
            System.out.println("❌ " + e.getMessage());
        }
    }

    private void viewRanking() {

        service.getCourses().forEach((id, course) ->
            System.out.println("  " + id + " - " + course.getCourseName()));
        System.out.print("请输入课程编号：");
        String courseId = scanner.nextLine().trim().toUpperCase();

            List<Map.Entry<Student, Double>> ranking = service.getCourseRanking(courseId);
            if (ranking.isEmpty()) {
                System.out.println("该课程暂无成绩数据");
                return;
            }
            String courseName = service.getCourses().get(courseId).getCourseName();
            System.out.println("\n【" + courseName + "】成绩排名：");
            System.out.println("─".repeat(35));
            for (int i = 0; i < ranking.size(); i++) {
                Map.Entry<Student, Double> entry = ranking.get(i);
                System.out.printf("第%2d名  %-8s  %.1f分%n",
                    i + 1,
                    entry.getKey().getName(),
                    entry.getValue());
            }
        } catch (IllegalArgumentException e) {
            System.out.println("❌ " + e.getMessage());
        }
    }
```
